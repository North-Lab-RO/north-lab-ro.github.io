## How it works

The LLM and the image model can't fit on a 24 GB card at the same time, so a GPU lease runs them in turn. Everything
is generated up front, which means playing a case needs no GPU at all.

## Solvability is a proof, not a hope

Every action a team can take (a search, an interview, a lab test, a records request) has prerequisites, reveals and
a clock cost. The validator computes everything reachable from the opening knowledge, then checks the cheapest path
to the proving evidence fits the time budget.

<CodeSnippet
  file="packages/shared/src/knowledge.ts"
  lang="ts"
  caption="A fixed-point closure over the case's actions. Any proving key outside it sends the case back to the stage that can actually produce it."
  code={`
export function computeClosure(
  initial: readonly KnowledgeKey[],
  actions: readonly DiscoveryAction[],
): Set<KnowledgeKey> {
  const held = new Set<KnowledgeKey>(initial);
  let grew = true;
  while (grew) {
    grew = false;
    for (const act of actions) {
      if (!isSatisfied(held, act.requires)) continue;
      for (const key of act.reveals) {
        if (!held.has(key)) {
          held.add(key);
          grew = true;
        }
      }
    }
  }
  return held;
}

// validation/solvability.ts
const closure = computeClosure(initial, actions);
const unreachable = draft.truth.provingEvidence.filter((key) => !closure.has(key));

const path = computeCriticalPath(initial, actions, draft.truth.provingEvidence);
const budgetSeconds = draft.profile.timeBudgetMinutes * 60;
const fraction = path.costSeconds / budgetSeconds;
if (fraction > draft.profile.maxCriticalPathFraction) {
  const used = Math.round(fraction * 100);
  const allowed = Math.round(draft.profile.maxCriticalPathFraction * 100);
  issues.push(issue(7,
    \`Critical path costs \${used}% of the time budget, above the \${allowed}% ceiling.\`));
}
`}
